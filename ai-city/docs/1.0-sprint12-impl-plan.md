```python
import json
import logging

import uuid
from typing import Any

from .config import settings
from .redis_pub import RedisPub

logger = logging.getLogger(__name__)


class ActionDispatcher:
    def __init__(self, redis: RedisPub) -> None:
        self.redis = redis

    async def say(

        npc_id: str,
        text: str,
        player_id: str | None = None,
        tile_id: str | None = None,
        options: list[dict] | None = None,

          "ts_ms": <int>,

            "npc_id": "npc_wang_boss_001",
            "player_id": "<uuid>",
            "tile_id": "tile_0_0",
            "say": "<台词文本>",
            "options": [{"id": "ask_business", "text": "..."}, ...]  // 仅剧本触发时
          }
        }
        """
        envelope = {
            "type": "npc_dialogue",
            "trace_id": str(uuid.uuid4()),
            "ts_ms": int(time.time() * 1000),
            "payload": {
                "npc_id": npc_id,
                "player_id": player_id or "",
                "tile_id": tile_id or "",
                "say": text,
                "options": options or [],
            },
        }
        try:
            await self.redis.publish(
                settings.REDIS_CHANNEL_NPC_DIALOGUE,
                json.dumps(envelope, ensure_ascii=False),
```

### redis_pub.py 新增 spec

参考 world-engine `apps/world-engine/src/redis_pub.rs` 模式（手写 RESP，不用 redis-py）：

```python
"""agent-os → Redis publisher（手写 RESP 协议）。

1.0 不引 redis-py；与 world-engine 同款。
"""
import asyncio
import logging
from dataclasses import dataclass, field

logger = logging.getLogger(__name__)


@dataclass
class RedisStats:
    messages_published: int = 0
    connect_errors: int = 0
    write_errors: int = 0
    flush_errors: int = 0

    def to_dict(self) -> dict:
        return {
            "messages_published": self.messages_published,
            "connect_errors": self.connect_errors,
            "write_errors": self.write_errors,
            "flush_errors": self.flush_errors,
        }


class RedisPub:
    """Fire-and-forget Redis publisher。"""

    def __init__(self, url: str) -> None:
        # url 形如 "redis://redis:6379/0" → 解析 host/port/db
        import urllib.parse
        parsed = urllib.parse.urlparse(url)
        self.host = parsed.hostname or "redis"
        self.port = parsed.port or 6379
        self.db = int(parsed.path.lstrip("/") or "0")
        self.stats = RedisStats()
        self._lock = asyncio.Lock()
        self._writer: asyncio.StreamWriter | None = None
        self._reader: asyncio.StreamReader | None = None

    async def _connect(self) -> bool:
        """单次连接；失败返 False，caller 决定是否 retry。"""
        async with self._lock:
            if self._writer and not self._writer.is_closing():
                return True
            try:
                self._reader, self._writer = await asyncio.open_connection(self.host, self.port)
                return True
            except Exception as e:
                self.stats.connect_errors += 1
                logger.warning(f"redis connect failed: {e}")
                return False

    async def publish(self, channel: str, payload: str) -> None:
        """PUBLISH channel payload。失败仅记 stats，不抛。"""
        if not await self._connect():
            return
        assert self._writer is not None
        # RESP: *3\r\n$7\r\nPUBLISH\r\n$<len-channel>\r\n<channel>\r\n$<len-payload>\r\n<payload>\r\n
        cmd = f"*3\r\n$7\r\nPUBLISH\r\n${len(channel)}\r\n{channel}\r\n${len(payload)}\r\n{payload}\r\n"
        try:
            self._writer.write(cmd.encode("utf-8"))
            await self._writer.drain()
            # 读 +OK（fire-and-forget，不消费响应也没事；读一下释放 buffer）
            self._stats_inc("messages_published")
        except Exception as e:
            self.stats.write_errors += 1
            logger.warning(f"redis publish failed: {e}")
            await self._reset()

    async def _reset(self) -> None:
        async with self._lock:
            if self._writer:
                try:
                    self._writer.close()
                    await self._writer.wait_closed()
                except Exception:
                    pass
            self._writer = None
            self._reader = None

    def _stats_inc(self, field_name: str) -> None:
        setattr(self.stats, field_name, getattr(self.stats, field_name) + 1)

    async def close(self) -> None:
        await self._reset()
```

**关键陷阱**：不引 `redis-py` 库 —— 1.0 简化，手写 RESP；2.0 才换 redis-py。

```ts
export const NPC_DIALOGUE_EVENT = 'aicity:npc_dialogue';

interface NpcDialoguePayload {
  npc_id: string;
  player_id: string;
  tile_id: string;
  say: string;
  options: Array<{ id: string; text: string }>;
}

// 既有 startWsBridge 函数内 WS message 处理分支加：
wsClient.onMessage((envelope: Envelope) => {
  if (envelope.type === 'npc_dialogue') {
    window.dispatchEvent(
      new CustomEvent(NPC_DIALOGUE_EVENT, { detail: envelope.payload as NpcDialoguePayload })
    );
  } else if (envelope.type === 'player_moved') {
    // 既有逻辑
    window.dispatchEvent(new CustomEvent(PLAYER_MOVED_EVENT, { detail: envelope.payload }));
  }
});
```

```go
package handlers

import (
    "net/http"

    "github.com/gin-gonic/gin"
    "github.com/aicity/api-gateway/internal/auth"
    "github.com/aicity/api-gateway/internal/npc"
)

type TalkRequest struct {
    Choice string `json:"choice"`
}

type TalkResponse struct {
    NpcID   string             `json:"npc_id"`
    Say     string             `json:"say"`
    Options []npc.NpcOption    `json:"options"`
}

type ErrorResponse struct {
    Error   string `json:"error"`
    TraceID string `json:"trace_id"`
}

func PostNPCTalk(loader *npc.TalkTreeLoader) gin.HandlerFunc {
    return func(c *gin.Context) {
        traceID := c.GetString("trace_id")

        // 1. 鉴权
        claims, ok := auth.ParseJWT(c)
        if !ok {
            c.JSON(http.StatusUnauthorized, ErrorResponse{Error: "no token", TraceID: traceID})
            return
        }
        playerID := claims.Subject

        // 2. 路径参数
        npcID := c.Param("id")
        if npcID == "" {
            c.JSON(http.StatusBadRequest, ErrorResponse{Error: "npc_id required", TraceID: traceID})
            return
        }

        // 3. 请求体
        var req TalkRequest
        if err := c.ShouldBindJSON(&req); err != nil || req.Choice == "" {
            c.JSON(http.StatusBadRequest, ErrorResponse{Error: "choice invalid", TraceID: traceID})
            return
        }

        // 4. 查 talk_tree
        reply, options, err := loader.ReplyTo(npcID, req.Choice)
        if err != nil {
            switch {
            case errors.Is(err, npc.ErrNPCNotFound):
                c.JSON(http.StatusNotFound, ErrorResponse{Error: "npc not found", TraceID: traceID})
            case errors.Is(err, npc.ErrChoiceInvalid):
                c.JSON(http.StatusBadRequest, ErrorResponse{Error: "choice not in talk_tree", TraceID: traceID})
            default:
                c.JSON(http.StatusInternalServerError, ErrorResponse{Error: "template read fail", TraceID: traceID})
            }
            return
        }

        _ = playerID  // 1.0 暂不持久化对话历史
        c.JSON(http.StatusOK, TalkResponse{
            NpcID:   npcID,
            Say:     reply,
            Options: options,
        })
    }
}
```

```python
"""Storyline 调度（1.0 简化版 = Saga 引擎占位）。"""
import asyncio
import logging
import random
import yaml
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Awaitable, Callable

logger = logging.getLogger(__name__)


@dataclass
class Storyline:
    id: str
    npc_id: str
    trigger: dict
    steps: list[dict]


class StorylineRegistry:
    def __init__(self, templates_dir: str, npc_registry, action_dispatcher) -> None:
        self.templates_dir = Path(templates_dir)
        self.npc_registry = npc_registry
        self.action_dispatcher = action_dispatcher
        self.storylines: dict[str, Storyline] = {}

    def load_all(self) -> None:
        """扫 templates_dir/*/storylines/*.yaml。"""
        storylines_dir = self.templates_dir / "wang_boss_storylines"
        if not storylines_dir.exists():
            return
        for f in storylines_dir.glob("*.yaml"):
            with open(f) as fp:
                data = yaml.safe_load(fp)
            self.storylines[data["id"]] = Storyline(
                id=data["id"],
                npc_id=data["npc_id"],
                trigger=data["trigger"],
                steps=data["steps"],
            )

    async def trigger(self, storyline_id: str, npc, player_id: str) -> None:
        if storyline_id not in self.storylines:
            logger.warning(f"storyline {storyline_id} not found")
            return
        storyline = self.storylines[storyline_id]
        logger.info(f"triggering storyline {storyline_id} for player {player_id}")
        for step in storyline.steps:
            await self._run_step(step, npc, player_id)

    async def _run_step(self, step: dict, npc, player_id: str) -> None:
        action = step.get("action")
        if action == "say":
            say_key = step["say_key"]
            npc_config = self.npc_registry.get(npc.agent_id)
            if not npc_config:
                return
            texts = npc_config.say.get(say_key, [])
            if not texts:
                return
            text = random.choice(texts) if step.get("random_pick") else texts[0]
            await self.action_dispatcher.say(
                npc_id=npc.agent_id,
                text=text,
                player_id=player_id,
                tile_id=npc.home_tile_id,
                options=None,  # 第 2 步再 show_options
            )
        elif action == "show_options":
            options_from = step.get("options_from", "")
            # 解析 "wang_boss.talk_tree.root.options"
            npc_config = self.npc_registry.get(npc.agent_id)
            if not npc_config or not npc_config.talk_tree:
                return
            root_node = npc_config.talk_tree.get("root", {})
            options_raw = root_node.get("options", [])  # [{id, text}, ...]
            await self.action_dispatcher.say(
                npc_id=npc.agent_id,
                text="",  # 第 1 步 say 已发过；这里只 push options
                player_id=player_id,
                tile_id=npc.home_tile_id,
                options=options_raw,
            )
```

```go
// steps.go
func runAcceptance(ctx context.Context, cfg *Config) int {
    // Step 0: 准备（login + WS connect + reset state）
    token, err := login(cfg)
    if err != nil { return failStep(0, "login", err) }

    wsConn, err := connectWS(cfg, token)
    if err != nil { return failStep(0, "ws connect", err) }
    defer wsConn.Close(websocket.StatusNormalClosure, "done")

    // Step 1: walk（玩家走 tile_0_0）

    }

    // Step 2: approach（玩家靠近王老板，|Δx|<100 ∧ |Δy|<100）
    if err := approach(cfg, token, "npc_wang_boss_001"); err != nil {
        return failStep(2, "approach", err)
    }

    // Step 3: NPC 主动 say（等 5s WS envelope）
    if err := waitForNPCSay(ctx, wsConn, "npc_wang_boss_001", 5*time.Second); err != nil {
        return failStep(3, "npc_say", err)
    }

    // Step 4: 玩家 reply（POST /v1/npc/:id/talk，验 reply + options）
    if err := reply(cfg, token, "npc_wang_boss_001", "ask_business"); err != nil {
        return failStep(4, "reply", err)

    return 0  // 5/5 PASS
}
```

### WS envelope 解析（envelope.go）

```go
type Envelope struct {
    Type    string          `json:"type"`
    TraceID string          `json:"trace_id"`
    TsMs    int64           `json:"ts_ms"`
    Payload json.RawMessage `json:"payload"`
}

// readEnvelope 在 5s 超时内读 1 个 envelope，按 type 过滤
func readEnvelope(conn *websocket.Conn, wantType string, timeout time.Duration) (*Envelope, error) {
    ctx, cancel := context.WithTimeout(context.Background(), timeout)
    defer cancel()
    for {
        var raw []byte
        if err := conn.Read(ctx); err != nil {
            return nil, err
        } else {
            raw = readBuf
        }
        var env Envelope
        if err := json.Unmarshal(raw, &env); err != nil {
            continue  // skip malformed
        }
        if env.Type == wantType {
            return &env, nil
        }
    }
}
```
