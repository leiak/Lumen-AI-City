### grpc.rs::move() 改动 spec

```rust
async fn move(&self, req: Request<MoveRequest>) -> Result<Response<MoveResponse>> {
    let entity_type = req.get_ref().entity_type();
    let entity_id = req.get_ref().entity_id.clone();
    match entity_type {
        EntityType::Player | EntityType::Unspecified => {
            // 现有 player 移动逻辑（不变）
            self.handle_player_move(req).await
        }
        EntityType::Npc => {
            // 新 NPC 移动逻辑
            self.handle_npc_move(req).await
        }
    }
}

async fn handle_npc_move(&self, req: Request<MoveRequest>) -> Result<Response<MoveResponse>> {
    let npc_id = req.get_ref().entity_id.clone();
    let target = req.get_ref().target.clone().unwrap();
    let mut npcs = self.npcs.lock().unwrap();
    let npc = npcs.get_mut(&npc_id)
        .ok_or_else(|| Status::not_found("npc not found"))?;
    if !npc.enabled {
        return Err(Status::failed_precondition("npc disabled"));
    }
    npc.x = target.x;
    npc.y = target.y;
    npc.tile_id = format!("tile_{}_{}", (target.x / 100.0) as i32, (target.y / 100.0) as i32);
    Ok(Response::new(MoveResponse {
        accepted: true,
        corrected_position: Some(target),
        sequence: req.get_ref().sequence,
        server_ts_ms: now_ms(),
    }))
}
```

### bt.py 改动 spec

**3 个节点类 + 1 个枚举**：
```python
from abc import ABC, abstractmethod
from dataclasses import dataclass
from enum import Enum
from typing import Callable, Awaitable


class NodeStatus(Enum):
    SUCCESS = "success"
    FAILURE = "failure"
    RUNNING = "running"


class BTNode(ABC):
    @abstractmethod
    async def tick(self, ctx: 'BTContext') -> NodeStatus: ...


@dataclass
class BTContext:
    """BT tick 上下文：NPC id + 当前状态 + 共享数据。"""
    npc_id: str
    blackboard: dict  # 跨节点共享数据
    dt: float = 0.0


class Sequence(BTNode):
    """按顺序执行子节点；任一 FAILURE 立即返回 FAILURE；全 SUCCESS 返 SUCCESS。"""
    def __init__(self, children: list[BTNode]) -> None:
        self.children = children
    async def tick(self, ctx: BTContext) -> NodeStatus:
        for child in self.children:
            status = await child.tick(ctx)
            if status != NodeStatus.SUCCESS:
                return status
        return NodeStatus.SUCCESS


class Selector(BTNode):
    """按顺序执行子节点；任一 SUCCESS 立即返回 SUCCESS；全 FAILURE 返 FAILURE。"""
    def __init__(self, children: list[BTNode]) -> None:
        self.children = children
    async def tick(self, ctx: BTContext) -> NodeStatus:
        for child in self.children:
            status = await child.tick(ctx)
            if status != NodeStatus.SUCCESS:
                return status
        return NodeStatus.SUCCESS


class Action(BTNode):
    """叶子节点：执行一个 async 函数，返 NodeStatus。"""
    def __init__(self, fn: Callable[[BTContext], Awaitable[NodeStatus]]) -> None:
        self.fn = fn
    async def tick(self, ctx: BTContext) -> NodeStatus:
        return await self.fn(ctx)
```

### 关键逻辑（描述）
- **3 节点 1.0 简化**：去 Inverter / Condition / Repeat 等（plan §六 拒绝）
- **异步节点**：所有 tick 都 async（与 agent-os loop 一致）
- **blackboard**：dict 共享数据（不引入 dataclass 简化）
- **返回枚举**：SUCCESS / FAILURE / RUNNING（1.0 不用 RUNNING，但留接口）

### 测试 spec（test_bt.py）

```python

@pytest.mark.asyncio
async def test_sequence_all_success():
    """Sequence: 3 子节点全 SUCCESS → 返 SUCCESS"""
    calls = []
    async def make_action(name, status):
```

```python
import os
import yaml
from dataclasses import dataclass, field
from pathlib import Path


@dataclass
class NPCConfig:
    npc_id: str
    agent_id: str
    name: str
    enabled: bool = False
    say_greeting: list[str] = field(default_factory=list)

class NPCRegistry:
    def __init__(self, templates_dir: str) -> None:
        self.templates_dir = Path(templates_dir)
        self.npcs: dict[str, NPCConfig] = {}  # agent_id → NPCConfig

    def load_all(self) -> None:
        """扫 templates_dir/*.yaml，加载所有 NPC。"""
        for f in self.templates_dir.glob("*.yaml"):
            with open(f) as fp:
                data = yaml.safe_load(fp)
            if not data.get("enabled", False):
                continue
            self.npcs[data["agent_id"]] = NPCConfig(
                npc_id=data["npc_id"],
                agent_id=data["agent_id"],
                name=data["name"],
                enabled=True,
                say_greeting=data.get("say", {}).get("greeting", []),
                home_tile_id=data.get("home_tile_id", ""),
                ocean=data.get("ocean", {}),
                bt_root=data.get("behavior_tree", ""),
            )

    def list_enabled(self) -> list[NPCConfig]:
        return [n for n in self.npcs.values() if n.enabled]

    def get(self, agent_id: str) -> NPCConfig | None:
        return self.npcs.get(agent_id)
```

```python
"""Agent OS 心跳循环。"""
import asyncio
import logging
import time
from agent_os.npc_registry import npc_registry
from agent_os.action_dispatcher import dispatcher, pick_greeting
from agent_os.player_listener import PlayerListener
from agent_os.world_client import WorldClient


logger = logging.getLogger(__name__)


class AgentOSLoop:
    """全 agent-os 心跳循环。"""

    def __init__(
        self,
        world_client: WorldClient,
        player_listener: PlayerListener,
        tick_seconds: float = 1.0,
    ) -> None:
        self.world_client = world_client
        self.player_listener = player_listener
        self.tick_interval = tick_seconds
        self._stop = asyncio.Event()

    async def run(self) -> None:
        logger.info("agent-os loop starting")
        while not self._stop.is_set():
            try:
                await self._tick()
            except Exception:
                logger.exception("tick error")
            await asyncio.sleep(self.tick_interval)

    async def _tick(self) -> None:
        """每个 tick：检查玩家 → 触发 enabled NPC 的 say。"""
        for npc in npc_registry.list_enabled():
            players_in_range = self.player_listener.players_in_tile(npc.home_tile_id)
            for player_pos in players_in_range:
                if not self.player_listener.is_welcomed(player_pos.player_id):
                    text = pick_greeting(npc)
                    await dispatcher.say(npc.npc_id, text)
                    self.player_listener.mark_welcomed(player_pos.player_id)

    async def stop(self) -> None:
        self._stop.set()
```

### 关键逻辑（描述）
- **loop 单实例**：不再每个 NPC 一个 AgentRuntime（plan §六 简化）
- **tick 1s**：listener 心跳 1 次/s 足够（1 demo = 1 玩家）
- **player_listener.players_in_tile**：返 [{player_id, x, y}]
- **mark_welcomed**：避免重复触发（Q3 决策）

### player_listener.py 改动 spec

```python
"""玩家位置 listener（内存版；2.0 接 Redis / PG）。"""
import time
from dataclasses import dataclass


@dataclass
class PlayerPosition:
    player_id: str
    x: float
    y: float
    ts_ms: int


class PlayerListener:
    def __init__(self) -> None:
        self.players: dict[str, PlayerPosition] = {}  # player_id → position
        self.welcomed_players: set[str] = set()        # 已 welcome 的 player_id

    def update_position(self, player_id: str, x: float, y: float) -> None:
        """外部（Redis 订阅 / gRPC stream）更新玩家位置。"""
        self.players[player_id] = PlayerPosition(player_id, x, y, int(time.time() * 1000))

    def players_in_tile(self, tile_id: str) -> list[PlayerPosition]:
        """返 tile 内的玩家（基于 tile_id 字符串匹配）。"""
        # tile_id 格式：tile_X_Y
        try:
            _, tx, ty = tile_id.split("_")
            tx, ty = int(tx), int(ty)
        except (ValueError, AttributeError):
            return []
        result = []
        for p in self.players.values():
            # tile 中心判定：玩家位置在 [tx*100, tx*100+100) × [ty*100, ty*100+100)
            px_tile = int(p.x // 100)
            py_tile = int(p.y // 100)
            if px_tile == tx and py_tile == ty:
                result.append(p)
        return result

    def is_in_range(self, player: PlayerPosition, npc_x: float, npc_y: float) -> bool:
        """Sprint 11 Q5 决策：同 tile（|Δx| < 100 ∧ |Δy| < 100）。"""
        return abs(player.x - npc_x) < 100 and abs(player.y - npc_y) < 100

    def is_welcomed(self, player_id: str) -> bool:
        return player_id in self.welcomed_players

    def mark_welcomed(self, player_id: str) -> None:
        self.welcomed_players.add(player_id)

    def clear_welcomed(self) -> None:
        """重启时调用（Q3 决策：内存丢）。"""
        self.welcomed_players.clear()


# 全局单例
player_listener = PlayerListener()
```
