# 2.0 阶段 3 B1 — 跨城 NPC 流式 实施 Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 2.0 阶段 2「同城内 NPC 流式」扩展到跨城场景 —— A 城玩家点 B 城 NPC 时，节拍包以流式跨过联邦边界，端到端体感接近同城。

**Architecture:** A 城玩家 → api-gateway A → a2a-gateway A（持 sid）→ mTLS gRPC server streaming → a2a-gateway B → agent-os B (LLM) → Redis B → a2a-gateway B（订阅）→ 流式回 A 城 → Redis A → ws-gateway A → 玩家。复用 stage 1 mTLS + stage 2 频道 + R_015 session 模型。

**Tech Stack:**
- a2a-gateway (Go) — 新增 `SayStreamForward` gRPC + HTTP SSE 入口 + 镜像 SessionStore
- api-gateway (Go) — `forwardCrossCityStream` SSE 转发 + 整段 fallback
- agent-os (Python) — 仅加 R_016/R_017 错误码；不动 dispatcher
- ws-gateway (Go) — **零改动**（已订阅 `aicity:npc:say_stream`）
- 测试: Go `testing` + miniredis + 双 docker compose stack smoke

```go
// Package crosscity session_mirror.go —— A 城 a2a-gateway 持 sid 的镜像 store。
//
// 镜像 agent-os SessionStore 语义（A2）：60min TTL + 重连补帧 buffer。
// 不复用 Python 源：避免跨语言依赖；Go 重写一份保持 API 一致。
package crosscity

import (
	"errors"
	"fmt"
	"sync"
	"time"

	"github.com/google/uuid"
)

// ErrSessionNotFound 跨城 session 缺失或过期（对应 R_015）。
var ErrSessionNotFound = errors.New("crosscity: session not found")

// MirrorBeat 镜像节拍包（不含 type / ts_ms / trace_id；上层补齐）。
type MirrorBeat struct {
	SentenceIdx int    `json:"sentence_idx"`
	Text        string `json:"text"`
	Emotion     string `json:"emotion"`
}

type mirrorSession struct {
	npcID      string
	traceID    string
	createdAt  time.Time
	buffer     []MirrorBeat
	done       bool
	complete   bool
}

// MirrorSessionStore 镜像 agent-os SessionStore（A 城 a2a-gateway 持有）。
type MirrorSessionStore struct {
	mu       sync.RWMutex
	sessions map[string]*mirrorSession
	ttl      time.Duration
}

func NewMirrorSessionStore(ttl time.Duration) *MirrorSessionStore {
	return &MirrorSessionStore{
		sessions: make(map[string]*mirrorSession),
		ttl:      ttl,
	}
}

// Create 新建跨城 session；返回 sid = "sess-<12hex>"（与 stage 2 同前缀）。
func (s *MirrorSessionStore) Create(npcID, traceID string) string {
	sid := fmt.Sprintf("sess-%s", uuid.NewString()[:12])
	s.mu.Lock()
	defer s.mu.Unlock()
	s.sessions[sid] = &mirrorSession{
		npcID:     npcID,
		traceID:   traceID,
		createdAt: time.Now(),
	}
	return sid
}

// Append 写入一帧节拍；sid 不存在或过期返回 error。
func (s *MirrorSessionStore) Append(sid string, idx int, text, emotion string) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	sess, ok := s.sessions[sid]
	if !ok || time.Since(sess.createdAt) > s.ttl {
		if ok {
			delete(s.sessions, sid)
		}
		return ErrSessionNotFound
	}
	sess.buffer = append(sess.buffer, MirrorBeat{
		SentenceIdx: idx, Text: text, Emotion: emotion,
	})
	return nil
}

// MarkDone 标记 session 流结束；complete=false 表示异常结束。
func (s *MirrorSessionStore) MarkDone(sid string, complete bool) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	sess, ok := s.sessions[sid]
	if !ok || time.Since(sess.createdAt) > s.ttl {
		return ErrSessionNotFound
	}
	sess.done = true
	sess.complete = complete
	return nil
}

// Buffer 返回 sentence_idx >= fromIdx 的所有帧。
func (s *MirrorSessionStore) Buffer(sid string, fromIdx int) ([]MirrorBeat, error) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	sess, ok := s.sessions[sid]
	if !ok || time.Since(sess.createdAt) > s.ttl {
		return nil, ErrSessionNotFound
	}
	out := make([]MirrorBeat, 0)
	for _, b := range sess.buffer {
		if b.SentenceIdx >= fromIdx {
			out = append(out, b)
		}
	}
	return out, nil
}

// IsDone 检查流是否结束（done 包已到达）。
func (s *MirrorSessionStore) IsDone(sid string) bool {
	s.mu.RLock()
	defer s.mu.RUnlock()
	sess, ok := s.sessions[sid]
	if !ok {
		return false
	}
	return sess.done
}

// NPCID 返回 session 关联的 NPC ID（重连补帧端点需要）。
func (s *MirrorSessionStore) NPCID(sid string) string {
	s.mu.RLock()
	defer s.mu.RUnlock()
	sess, ok := s.sessions[sid]
	if !ok {
		return ""
	}
	return sess.npcID
}
```

```go
// 调用由 T06 补全（依赖 dispatcher client）。
func (s *Service) SayStreamForward(stream a2av1.A2AGateway_SayStreamForwardServer) error {
	ctx := stream.Context()
	first, err := stream.Recv()
	if err != nil {
		return err
	}
	init := first.GetInit()
	if init == nil {
		return status.Error(codes.InvalidArgument, "first frame must be SayRequestInit")
	}
	if init.NpcID == "" {
		return status.Error(codes.InvalidArgument, "R_009:npc_id required")
	}
	log.Printf("a2a-gateway: SayStreamForward start npc_id=%s session=%s trace=%s",
		init.NpcID, init.SessionID, init.TraceID)

	// T06 实现：订阅 B 城 Redis + 转发回 A 城
	// 当前骨架：直接返 done(complete=true) 让 A 城测试流程跑通。

		Type:       "npc_say_stream_done",
		NpcID:      init.NpcID,
		SessionID:  init.SessionID,
		Complete:   boolPtr(true),
		SentenceCount: int32Ptr(0),
		TsMs:       time.Now().UnixMilli(),
		TraceID:    init.TraceID,
	})
}
```

```go
package httpgw

import (
	"encoding/json"
	"net/http"
	"time"

	"github.com/aicity/a2a-gateway/internal/crosscity"
	"github.com/gin-gonic/gin"
)

// SayStreamRequest POST /v1/federation/say_stream 请求体。
type SayStreamRequest struct {
	NpcID       string             `json:"npc_id"`
	PlayerID    string             `json:"player_id"`
	PlayerInput string             `json:"player_input"`
	SessionID   string             `json:"session_id"` // 可选；缺省生成
	TraceID     string             `json:"trace_id"`
	Context     []ContextMsgEntry  `json:"context"`
}

type ContextMsgEntry struct {
	Role    string `json:"role"`
	Content string `json:"content"`
}

// SayStreamHandler 处理 A 城 api-gateway POST /v1/federation/say_stream。
//
// 流程（初版骨架）：
//  1. 校验 npc_id 非空（R_009）；
//  2. 在 MirrorStore 创建/继承 session；
//  3. 返 SSE 头 + 占位 done 包（框架内，回包完整 T06 接入 B 城 gRPC）。
func SayStreamHandler(store *crosscity.MirrorSessionStore, router CrossCityRouter, redisPub RedisPublisher) gin.HandlerFunc {
	return func(c *gin.Context) {
		var req SayStreamRequest
		if err := c.ShouldBindJSON(&req); err != nil {
			c.JSON(http.StatusBadRequest, gin.H{"code": "R_009", "message": err.Error()})
			return
		}
		if req.NpcID == "" {
			c.JSON(http.StatusBadRequest, gin.H{"code": "R_009", "message": "npc_id required"})
			return
		}

		// session 创建（A 城持有）
		if req.SessionID == "" {
			req.SessionID = store.Create(req.NpcID, req.TraceID)
		}
		if req.TraceID == "" {
			req.TraceID = "tr-" + req.SessionID[len("sess-"):]
		}

		// SSE 响应头
		c.Writer.Header().Set("Content-Type", "text/event-stream")
		c.Writer.Header().Set("Cache-Control", "no-cache")
		c.Writer.Header().Set("Connection", "keep-alive")
		c.Writer.Header().Set("X-Accel-Buffering", "no")
		c.Writer.Flush()

		// init 帧
		initPayload := map[string]any{
			"type":       "init",
			"session_id": req.SessionID,
			"trace_id":   req.TraceID,
			"ts_ms":      time.Now().UnixMilli(),
		}
		writeSSE(c.Writer, initPayload)
```

```go
	ctx := stream.Context()
	first, err := stream.Recv()
	if err != nil {
		return err
	}
	init := first.GetInit()
	if init == nil {
		return status.Error(codes.InvalidArgument, "first frame must be SayRequestInit")
	}

		return status.Error(codes.InvalidArgument, "R_009:npc_id required")

	// T06.5 补；本任务只占位）
	go s.invokeAgentOS(ctx, init)

	// 订阅本城 Redis aicity:npc:say_stream，按 session_id 过滤
	sub := s.Redis.Subscribe(ctx, "aicity:npc:say_stream")
	defer sub.Close()
	ch := sub.Channel()

	deadline := time.NewTimer(30 * time.Second)
	defer deadline.Stop()

	for {
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-deadline.C:
			return status.Error(codes.DeadlineExceeded, "R_017:stream timeout")
		case msg, ok := <-ch:
			if !ok {
				return status.Error(codes.Unavailable, "R_016:redis channel closed")
			}
			var payload map[string]any
			if err := json.Unmarshal([]byte(msg.Payload), &payload); err != nil {

			}

				continue
			}

			if payload["type"] == "npc_say_stream_done" {

		}
	}
```

```go
// SessionBufferHandler GET /v1/federation/sessions/:sid/buffer?from_idx=N
//
// 返回从 sentence_idx >= N 的所有帧（镜像 stage 2 SessionStore 语义）。
// sid 不存在或过期 → 404 R_015。
func SessionBufferHandler(store *crosscity.MirrorSessionStore) gin.HandlerFunc {
	return func(c *gin.Context) {
		sid := c.Param("sid")
		fromIdx, _ := strconv.Atoi(c.DefaultQuery("from_idx", "0"))

		npcID := store.NPCID(sid)
		if npcID == "" {
			c.JSON(http.StatusNotFound, gin.H{
				"code": "R_015", "message": "session 不存在或已过期",
			})
			return
		}

		buf, err := store.Buffer(sid, fromIdx)
		if err != nil {
			if errors.Is(err, crosscity.ErrSessionNotFound) {
				c.JSON(http.StatusNotFound, gin.H{
					"code": "R_015", "message": "session 不存在或已过期",
				})
				return
			}
			c.JSON(http.StatusBadRequest, gin.H{"code": "R_009", "message": err.Error()})
			return
		}

		c.JSON(http.StatusOK, gin.H{
			"session_id": sid,
			"npc_id":     npcID,
			"complete":   store.IsDone(sid),
			"buffer":     buf,
		})
```

`apps/a2a-gateway/cmd/acceptance_2_2/main.go`：

```go
// acceptance_2_2 —— B1 跨城流式 5 步 E2E 验证。
//
// 步骤：
//  1. A 城登录 → token
//  2. 跨城 NPC 流式触发 → SSE 收到 init + done 帧
//  3. 验证 B 城 NPC 收到 npc_say_stream 包（订阅 Redis）
//  4. 验证 emotion ∈ 8 类
//  5. 重连补帧 GET /v1/federation/sessions/{sid}/buffer
package main

import (
	"bufio"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"os"
	"strings"
	"time"

	"github.com/redis/go-redis/v9"
)

const (
	apiBase  = "http://api-gateway:8080"
	a2aBase  = "http://a2a-gateway:8083"
	player   = "demo"
	password = "demo123"
	npcID    = "npc_b_grace_healer_001" // B 城 NPC
)

func main() {
	ctx, cancel := context.WithTimeout(context.Background(), 90*time.Second)
	defer cancel()

	rdb := redis.NewClient(&redis.Options{Addr: "redis:6379"})

	// Step 1: login
	fmt.Println("Step 1: A 城登录")
	token, err := login(ctx)
	if err != nil {
		die("step 1", err)
	}
	fmt.Printf("  PASS login → token=%s...\n", token[:8])

	// Step 2: SSE 触发
	fmt.Println("Step 2: SSE 触发跨城 NPC 流式")
	sid, err := triggerSSE(ctx, token)
	if err != nil {
		die("step 2", err)
	}
	fmt.Printf("  PASS SSE → sid=%s\n", sid)

	// Step 3: 订阅 Redis 验证 beats
	fmt.Println("Step 3: 验证 aicity:npc:say_stream 节拍包")
	beats, err := subscribeBeats(ctx, rdb, sid, 8)
	if err != nil || len(beats) == 0 {
		die("step 3", fmt.Errorf("beats=%d err=%v", len(beats), err))
	}
	fmt.Printf("  PASS 收到 %d beats\n", len(beats))

	// Step 4: emotion ∈ 8 类
	fmt.Println("Step 4: emotion ∈ 8 类验证")
	allowed := map[string]bool{
		"happy": true, "sad": true, "angry": true, "surprised": true,
		"thinking": true, "embarrassed": true, "curious": true, "neutral": true,
	}
	for _, b := range beats {
		emo, _ := b["emotion"].(string)
		if emo != "" && !allowed[emo] {
			die("step 4", fmt.Errorf("invalid emotion %q", emo))
		}
	}
	fmt.Println("  PASS all emotions ∈ 8 classes")

	// Step 5: 重连补帧
	fmt.Println("Step 5: 重连补帧 GET /v1/federation/sessions/{sid}/buffer")
	buf, err := fetchBuffer(ctx, sid)
	if err != nil {
		die("step 5", err)
	}
	fmt.Printf("  PASS buffer=%d frames\n", len(buf))

	fmt.Println("\n=== acceptance_2_2 5/5 PASS ===")
}

func login(ctx context.Context) (string, error) {
	body, _ := json.Marshal(map[string]string{"username": player, "password": password})
	req, _ := http.NewRequestWithContext(ctx, http.MethodPost,
		apiBase+"/v1/auth/login", strings.NewReader(string(body)))
	req.Header.Set("Content-Type", "application/json")
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()
	var out struct{ Token string `json:"token"` }
	json.NewDecoder(resp.Body).Decode(&out)
	return out.Token, nil
}

func triggerSSE(ctx context.Context, token string) (string, error) {
	body, _ := json.Marshal(map[string]any{
		"npc_id":       npcID,
		"player_id":    "demo-player",
		"player_input": "你好",
	})
	req, _ := http.NewRequestWithContext(ctx, http.MethodPost,
		a2aBase+"/v1/federation/say_stream", strings.NewReader(string(body)))
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", "Bearer "+token)
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()

	scanner := bufio.NewScanner(resp.Body)
	var sid string
	for scanner.Scan() {
		line := scanner.Text()
		if !strings.HasPrefix(line, "data: ") {
			continue
		}
		var payload map[string]any
		json.Unmarshal([]byte(strings.TrimPrefix(line, "data: ")), &payload)
		if t, _ := payload["type"].(string); t == "init" {
			sid, _ = payload["session_id"].(string)
		}
	}
	return sid, nil
}

func subscribeBeats(ctx context.Context, rdb *redis.Client, sid string, seconds int) ([]map[string]any, error) {
	sub := rdb.Subscribe(ctx, "aicity:npc:say_stream")
	defer sub.Close()
	if _, err := sub.Receive(ctx); err != nil {
		return nil, err
	}
	ch := sub.Channel()
	deadline := time.After(time.Duration(seconds) * time.Second)
	var beats []map[string]any
	for {
		select {
		case msg := <-ch:
			var payload map[string]any
			json.Unmarshal([]byte(msg.Payload), &payload)
			if payload["session_id"] == sid {
				beats = append(beats, payload)
			}
		case <-deadline:
			return beats, nil
		}
	}
}

func fetchBuffer(ctx context.Context, sid string) ([]map[string]any, error) {
	resp, err := http.Get(a2aBase + "/v1/federation/sessions/" + sid + "/buffer?from_idx=0")
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	if resp.StatusCode != 200 {
		return nil, fmt.Errorf("status %d", resp.StatusCode)
	}
	var out struct {
		Buffer []map[string]any `json:"buffer"`
	}
	body, _ := io.ReadAll(resp.Body)
	json.Unmarshal(body, &out)
	return out.Buffer, nil
}

func die(step string, err error) {
	fmt.Fprintf(os.Stderr, "FAIL %s: %v\n", step, err)
	os.Exit(1)
}
```
