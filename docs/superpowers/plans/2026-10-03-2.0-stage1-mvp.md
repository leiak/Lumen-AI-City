- [ ] **Step 1: 写 CA 生成脚本**

`ai-city/scripts/gen-ca.sh`：

```bash
#!/bin/bash
set -euo pipefail
CN="${1:-aicity-ca}"
DIR="certs"
mkdir -p "$DIR"
openssl genrsa -out "$DIR/$CN.key" 2048
openssl req -new -x509 -days 365 -key "$DIR/$CN.key" \
  -out "$DIR/$CN.crt" -subj "/CN=$CN/O=AI-City/C=CN"
echo "CA cert: $DIR/$CN.crt"
echo "CA key:  $DIR/$CN.key"
```

- [ ] **Step 2: 写证书签发脚本**

`ai-city/scripts/issue-cert.sh`：

```bash
#!/bin/bash
set -euo pipefail
SERVICE="${1:?usage: issue-cert.sh <service-name> <city-or-role>}"
ROLE="${2:?usage: issue-cert.sh <service-name> <city-or-role>}"
CA_CN="aicity-ca"
DIR="certs"
DAYS=90

# Generate key + CSR
openssl genrsa -out "$DIR/$SERVICE.key" 2048
cat > "$DIR/$SERVICE.cnf" <<EOF
[req]
distinguished_name = req
req_extensions = v3_req
prompt = no
[req_distinguished_name]
CN = $SERVICE
[v3_req]
keyUsage = critical, digitalSignature, keyEncipherment
extendedKeyUsage = serverAuth, clientAuth
subjectAltName = @alt_names
[alt_names]
DNS.1 = $SERVICE
DNS.2 = localhost
IP.1 = 127.0.0.1
EOF

openssl req -new -key "$DIR/$SERVICE.key" \
  -out "$DIR/$SERVICE.csr" -config "$DIR/$SERVICE.cnf"

# Sign with CA
openssl x509 -req -in "$DIR/$SERVICE.csr" \
  -CA "$DIR/$CA_CN.crt" -CAkey "$DIR/$CA_CN.key" -CAcreateserial \
  -out "$DIR/$SERVICE.crt" -days $DAYS \
  -extensions v3_req -extfile "$DIR/$SERVICE.cnf"

# Print fingerprint
openssl x509 -in "$DIR/$SERVICE.crt" -noout -fingerprint -sha256 | \
  awk -F'=' '{print $2}' | tr -d ':' > "$DIR/$SERVICE.fingerprint"

echo "Cert: $DIR/$SERVICE.crt (valid $DAYS days)"
echo "Fingerprint: $(cat $DIR/$SERVICE.fingerprint)"
```

-- 2.0 跨城联邦路由表
CREATE TABLE IF NOT EXISTS a2a_npc_routes (
  npc_id TEXT PRIMARY KEY,
  city TEXT NOT NULL,
  world_engine_addr TEXT NOT NULL,
  grpc_cert_fingerprint TEXT NOT NULL,
  enabled BOOLEAN DEFAULT TRUE,
  last_heartbeat_ms BIGINT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_a2a_npc_routes_city ON a2a_npc_routes(city);

-- NPC 心跳表（每 30s 上报）
CREATE TABLE IF NOT EXISTS a2a_npc_heartbeats (
  npc_id TEXT PRIMARY KEY REFERENCES a2a_npc_routes(npc_id),
  last_seen_ms BIGINT NOT NULL,
  cpu_percent REAL,
  memory_mb INTEGER
);

-- 短时记忆 player_session 表（PG fallback）
CREATE TABLE IF NOT EXISTS memory_player_session (
  id SERIAL PRIMARY KEY,
  npc_id TEXT NOT NULL,
  player_id TEXT NOT NULL,
  message JSONB NOT NULL,
  embedding REAL[],  -- 1024 维
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_memory_player_session_npc_player
  ON memory_player_session(npc_id, player_id, created_at DESC);

# ai-city/apps/agent-os/src/agent_os/llm/rules.py
from dataclasses import dataclass

@dataclass
class RuleCheckResult:
    allow_continue: bool = True
    forced_closing: str = ""
    exceeded: bool = False
    reason: str = ""

class ChatTurnRule:
    """§10 规则 1: chat_turns=6 强制收尾"""
    def __init__(self, max_turns: int = 6):
        self.max_turns = max_turns

    def check(self, turn: int, last_response: str) -> RuleCheckResult:
        if turn >= self.max_turns:
            return RuleCheckResult(
                allow_continue=False,
                forced_closing="闲聊到此，下次再来！",
            )
        return RuleCheckResult(allow_continue=True)

class TokenCapRule:
    """§10 规则 2: token 上限"""
    def __init__(self, max_input_tokens: int = 4000, max_output_tokens: int = 300):
        self.max_input_tokens = max_input_tokens
        self.max_output_tokens = max_output_tokens

    def check(self, input_tokens: int, output_tokens: int) -> RuleCheckResult:
        if input_tokens > self.max_input_tokens:
            return RuleCheckResult(exceeded=True, reason=f"input_tokens {input_tokens} > {self.max_input_tokens}")
        if output_tokens > self.max_output_tokens:
            return RuleCheckResult(exceeded=True, reason=f"output_tokens {output_tokens} > {self.max_output_tokens}")
        return RuleCheckResult()

def enforce_chat_turn(turn: int, last_response: str, max_turns: int = 6) -> str | None:
    """返回 None 表示继续；返回字符串表示强制收尾台词"""
    rule = ChatTurnRule(max_turns)
    result = rule.check(turn, last_response)
    if not result.allow_continue:
        return result.forced_closing
    return None
```

from .llm.short_circuit import is_short_circuit

    text: str
    ts_ms: int
    trace_id: str
    chat_turn: int = 0
    forced_close: bool = False

class ActionDispatcher:
    def __init__(self, llm_client: Optional[LLMProvider] = None):
        self.llm_client = llm_client

    async def say(self, npc_id: str, player_input: str, npc_context: list) -> DispatcherSayResult:

            npc_id=npc_id,

            ts_ms=int(time.time() * 1000),
            trace_id=str(uuid.uuid4()),
        )

    async def say_with_llm(

        npc_id: str,
        player_input: str,
        npc_context: List[Dict[str, str]],

                npc_id=npc_id, text="你好！", ts_ms=int(time.time()*1000),
                trace_id=str(uuid.uuid4()), chat_turn=chat_turn,
            )
        # chat_turn=6 强制收尾
        closing = enforce_chat_turn(chat_turn, "")
        if closing:
            return DispatcherSayResult(
                npc_id=npc_id, text=closing, ts_ms=int(time.time()*1000),
                trace_id=str(uuid.uuid4()), chat_turn=chat_turn,
                forced_close=True,
            )
        # 提取 NPC slug from npc_<city>_<slug>

        resp = await self.llm_client.complete(req)
        return DispatcherSayResult(
            npc_id=npc_id, text=resp.text, ts_ms=int(time.time()*1000),

import os

# Claude Sonnet 4.6 pricing (per 1M tokens)
INPUT_COST = 3.0
OUTPUT_COST = 15.0
MONTHLY_BUDGET_USD = float(os.getenv("LLM_MONTHLY_BUDGET_USD", "100"))

class CostTracker:
    def __init__(self):
        self.total_input_tokens = 0
        self.total_output_tokens = 0

    def record(self, input_tokens: int, output_tokens: int):
        self.total_input_tokens += input_tokens
        self.total_output_tokens += output_tokens

    def cost_usd(self) -> float:
        return (
            self.total_input_tokens / 1e6 * INPUT_COST +
            self.total_output_tokens / 1e6 * OUTPUT_COST
        )

    def over_budget(self) -> bool:
        return self.cost_usd() > MONTHLY_BUDGET_USD

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List
from .milvus_client import get_milvus_client
from .embeddings import embed_text

app = FastAPI(title="memory-service", version="2.0.0")

class RecallRequest(BaseModel):
    npc_id: str
    player_id: str
    query: str
    top_k: int = 5

class RecallResponse(BaseModel):
    messages: List[str]
    scores: List[float]

@app.post("/v1/recall", response_model=RecallResponse)
async def recall(req: RecallRequest):
    collection_name = f"mem_{req.npc_id.split('_')[1]}_{'_'.join(req.npc_id.split('_')[2:])}"
    client = get_milvus_client()
    try:
        collection = client.Collection(collection_name)
    except Exception:
        # 降级到 PG short-term
        return await recall_from_pg(req)
    collection.load()
    query_embedding = embed_text(req.query)
    results = collection.search(
        data=[query_embedding],
        anns_field="embedding",
        param={"metric_type": "COSINE", "params": {"nprobe": 10}},
        limit=req.top_k,
        expr=f'player_id == "{req.player_id}"',
        output_fields=["text"],
    )
    messages = [hit.entity.get("text") for hit in results[0]]
    scores = [hit.score for hit in results[0]]
    return RecallResponse(messages=messages, scores=scores)

        host=os.getenv("POSTGRES_HOST", "postgres"),

async def recall_from_pg(req):
    """Milvus 不可达时降级到 PG short-term"""
    with get_pg_conn() as conn:
        with conn.cursor() as cur:
            cur.execute("""
                SELECT message->>'text', 1.0 - (embedding <=> %s::real[]) AS score
                FROM memory_player_session
                WHERE npc_id = %s AND player_id = %s
                ORDER BY created_at DESC LIMIT %s
            """, (embed_text(req.query), req.npc_id, req.player_id, req.top_k))
            rows = cur.fetchall()
    if not rows:
        return RecallResponse(messages=[], scores=[])
    return RecallResponse(messages=[r[0] for r in rows], scores=[r[1] for r in rows])

#!/usr/bin/env python3
"""Milvus recall top-5 准确率离线评估"""
import json
import requests
from collections import defaultdict

EVAL_SET = "data/milvus-eval-set.jsonl"
MEMORY_URL = "http://localhost:9200"

def load_eval_set():
    items = []
    with open(EVAL_SET) as f:
        for line in f:
            items.append(json.loads(line))
    return items

def evaluate():
    items = load_eval_set()
    hits = defaultdict(lambda: {"correct": 0, "total": 0})
    for item in items:
        resp = requests.post(f"{MEMORY_URL}/v1/recall", json={
            "npc_id": item["npc_id"],
            "player_id": item["player_id"],
            "query": item["query"],
            "top_k": 5,
        })
        recalled = set(resp.json()["messages"])
        expected = set(item["expected_messages"])
        if expected & recalled:
            hits[item["npc_id"]]["correct"] += 1
        hits[item["npc_id"]]["total"] += 1
    print(f"{'NPC':<25} {'Recall@5':<10} {'Samples'}")
    overall_correct = overall_total = 0
    for npc_id, stat in hits.items():
        acc = stat["correct"] / stat["total"] if stat["total"] else 0
        print(f"{npc_id:<25} {acc:<10.2%} {stat['total']}")
        overall_correct += stat["correct"]
        overall_total += stat["total"]
    overall = overall_correct / overall_total if overall_total else 0
    print(f"\n{'OVERALL':<25} {overall:<10.2%} {overall_total}")
    return overall >= 0.70

if __name__ == "__main__":
    import sys
    sys.exit(0 if evaluate() else 1)

package state

import (
    "encoding/json"
    "fmt"
    "sync"
    "time"
)

type SagaStatus string
const (
    StatusPending     SagaStatus = "PENDING"
    StatusRunning     SagaStatus = "RUNNING"
    StatusSuccess     SagaStatus = "SUCCESS"
    StatusCompensated SagaStatus = "COMPENSATED"
    StatusFailed      SagaStatus = "FAILED"
)

type SagaStep struct {
    WorkerID  string                 `json:"worker_id"`
    Action    string                 `json:"action"`
    Payload   map[string]any         `json:"payload"`
    Status    SagaStatus             `json:"status"`
    StartedAt int64                  `json:"started_at_ms"`
    EndedAt   int64                  `json:"ended_at_ms"`
    Error     string                 `json:"error,omitempty"`
}

type Saga struct {
    ID         string     `json:"saga_id"`
    PlayerID   string     `json:"player_id"`
    Steps      []SagaStep `json:"steps"`
    Status     SagaStatus `json:"status"`
    CreatedAt  int64      `json:"created_at_ms"`
    UpdatedAt  int64      `json:"updated_at_ms"`
    Compensated bool      `json:"compensated"`
}

type StateMachine struct {
    mu    sync.RWMutex
    sagas map[string]*Saga
}

func NewStateMachine() *StateMachine {
    return &StateMachine{sagas: make(map[string]*Saga)}
}

func (sm *StateMachine) CreateSaga(id, playerID string, steps []SagaStep) *Saga {
    sm.mu.Lock()
    defer sm.mu.Unlock()
    saga := &Saga{
        ID: id, PlayerID: playerID, Steps: steps,
        Status: StatusPending, CreatedAt: time.Now().UnixMilli(),
    }
    sm.sagas[id] = saga
    return saga
}

func (sm *StateMachine) UpdateStep(sagaID, workerID string, status SagaStatus, err string) error {
    sm.mu.Lock()
    defer sm.mu.Unlock()
    saga, ok := sm.sagas[sagaID]
    if !ok { return fmt.Errorf("saga %s not found", sagaID) }
    for i := range saga.Steps {
        if saga.Steps[i].WorkerID == workerID {
            saga.Steps[i].Status = status
            saga.Steps[i].Error = err
            saga.Steps[i].EndedAt = time.Now().UnixMilli()
            break
        }
    }
    saga.UpdatedAt = time.Now().UnixMilli()
    return nil
}
```

```go
// apps/a2a-gateway/internal/router/router.go
package router

import (
    "database/sql"
    "sync"
    "time"
)

type NPCRoute struct {
    NPCID               string
    City                string
    WorldEngineAddr     string
    GrpcCertFingerprint string
    Enabled             bool
    LastHeartbeatMs     int64
}

type Router struct {
    mu     sync.RWMutex
    routes map[string]*NPCRoute
}

func New(db *sql.DB) (*Router, error) {
    r := &Router{routes: make(map[string]*NPCRoute)}
    if err := r.loadFromDB(db); err != nil { return nil, err }
    go r.heartbeatChecker(db)
    return r, nil
}

func (r *Router) loadFromDB(db *sql.DB) error {
    rows, err := db.Query(`SELECT npc_id, city, world_engine_addr, grpc_cert_fingerprint, enabled, last_heartbeat_ms FROM a2a_npc_routes`)
    if err != nil { return err }
    defer rows.Close()
    r.mu.Lock()
    defer r.mu.Unlock()
    for rows.Next() {
        rt := &NPCRoute{}
        if err := rows.Scan(&rt.NPCID, &rt.City, &rt.WorldEngineAddr, &rt.GrpcCertFingerprint, &rt.Enabled, &rt.LastHeartbeatMs); err != nil {
            return err
        }
        r.routes[rt.NPCID] = rt
    }
    return nil
}

func (r *Router) Lookup(npcID string) (*NPCRoute, bool) {
    r.mu.RLock()
    defer r.mu.RUnlock()
    rt, ok := r.routes[npcID]
    if !ok || !rt.Enabled { return nil, false }
    if time.Now().UnixMilli() - rt.LastHeartbeatMs > 90000 {
        return nil, false  // 离线
    }
    return rt, true
}

func (r *Router) heartbeatChecker(db *sql.DB) {
    ticker := time.NewTicker(30 * time.Second)
    for range ticker.C {
        // 标记超过 90s 未心跳的 route 为 disabled
        _, _ = db.Exec(`UPDATE a2a_npc_routes SET enabled = FALSE WHERE last_heartbeat_ms < $1`, time.Now().UnixMilli() - 90000)
        _ = r.loadFromDB(db)
    }
}
```

package crosscity

import (
    "crypto/tls"
    "crypto/x509"
    "google.golang.org/grpc"
    "google.golang.org/grpc/credentials"
    "os"
)

func newTLSConfig() (*tls.Config, error) {
    cert, err := tls.LoadX509KeyPair(os.Getenv("TLS_CERT"), os.Getenv("TLS_KEY"))
    if err != nil { return nil, err }
    caBytes, err := os.ReadFile(os.Getenv("TLS_CA"))
    if err != nil { return nil, err }
    pool := x509.NewCertPool()
    pool.AppendCertsFromPEM(caBytes)
    return &tls.Config{
        Certificates: []tls.Certificate{cert},
        RootCAs:      pool,
        MinVersion:   tls.VersionTLS13,
    }, nil
}

func Dial(addr string) (*grpc.ClientConn, error) {
    tlsConfig, _ := newTLSConfig()
    return grpc.Dial(addr, grpc.WithTransportCredentials(credentials.NewTLS(tlsConfig)))
}
